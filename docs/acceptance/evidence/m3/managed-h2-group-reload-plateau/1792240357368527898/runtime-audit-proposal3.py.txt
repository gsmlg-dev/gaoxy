class AuditError(Exception):
    pass

def need(ok, label):
    if not ok:
        raise AuditError(label)

def exact(obj, keys, label):
    need(type(obj) is dict and set(obj) == set(keys), label + '.exact_keys')

def parse_json(raw):
    def pairs(rows):
        out = {}
        for key, value in rows:
            need(key not in out, 'json.duplicate_key')
            out[key] = value
        return out
    def invalid_constant(value):
        raise AuditError('json.nonfinite_constant')
    return json.loads(raw, object_pairs_hook=pairs, parse_constant=invalid_constant)

def integer(n, low=0, high=2**63-1):
    return type(n) is int and low <= n <= high

def positive(n):
    return integer(n, 1)

def inspect_id(value):
    need(type(value) is dict and set(value) == {'inspect'} and
         isinstance(value['inspect'], str) and 0 < len(value['inspect']) <= 262144,
         'identity.inspect_shell')
    return value['inspect']

def opaque(value, kind):
    exact(value, ('opaque', 'identity'), 'opaque')
    need(value['opaque'] == kind and integer(value['identity'], 1, 512), 'opaque.kind_or_id')
    return (kind, value['identity'])

def atom(value, expected):
    need(inspect_id(value) == ':' + expected, 'atom.' + expected)

def elixir_field(value, field):
    # Closed existing inspect ID shell; parse only the requested scalar field.
    text = inspect_id(value)
    match = re.search(r'\b' + re.escape(field) + r': (#[A-Za-z]+<[^>]+>|"[^"\\]*")', text)
    need(match is not None, 'inspect.scalar_field.' + field)
    return match.group(1)

def canonical(value):
    return json.dumps(value, sort_keys=True, separators=(',', ':'))

def bounded(path, cap):
    # The auditor must be invoked only on the explicit safe command files.
    name = Path(path).name.lower()
    need('netlog' not in name and 'raw-terminal' not in name and 'private' not in name and
         not name.endswith('.pem'), 'input.safe_named_file')
    with open(path, 'rb') as stream:
        raw = stream.read(cap + 1)
    need(len(raw) <= cap, 'input.bound')
    return raw

def inventory(inv, sut, previous=None):
    exact(inv, ('sut', 'generation', 'group', 'transports', 'creators', 'baseline',
                'group_member_configs', 'group_members_absolute_times', 'transport_controls',
                'group_pin_deadlines', 'cumulative'), 'inventory')
    need(inv['sut'] == sut, 'identity.sut_stable')
    need(len(inv['transports']) == len(inv['creators']) == len(inv['transport_controls']) == 2,
         'inventory.two_transports')
    controls = inv['transport_controls']
    need(len({inspect_id(c['resource']) for c in controls}) == 2, 'inventory.distinct_resources')
    for handle, c in zip(inv['transports'], controls):
        need(elixir_field(handle, 'resource') == inspect_id(c['resource']), 'inventory.handle_resource')
    if previous:
        need(inv['generation'] != previous['generation'] and inv['group'] != previous['group'], 'identity.fresh_generation_group')
        old = {inspect_id(c['resource']) for c in previous['transport_controls']}
        need(not old.intersection(inspect_id(c['resource']) for c in controls), 'identity.fresh_resources')
    return controls

def peer_setup(setup, upstream):
    peers = setup['peers']
    need(len(peers) == 6, 'setup.six_peers')
    by_key = {(p['epoch'], p['lane']): p for p in peers}
    need(set(by_key) == set(itertools.product(range(3), ('a', 'b'))), 'setup.peer_epochs')
    need(len({inspect_id(p['peer']) for p in peers}) == 6, 'setup.distinct_peers')
    need(by_key[(0, 'a')]['port'] != by_key[(0, 'b')]['port'], 'setup.distinct_lane_ports')
    for (epoch, lane), p in by_key.items():
        need(inspect_id(p['address']) == '{127, 0, 0, ' + str(epoch + 1) + '}', 'setup.address')
        need(p['port'] == by_key[(0, lane)]['port'] and integer(p['port'], 1, 65535), 'setup.fixed_port')
        need(p['source_class'] == setup['classes'][0 if lane == 'a' else 1], 'setup.lane_class')
        need(type(p['observed_setup_to_ready_ms']) is int and p['observed_setup_to_ready_ms'] >= 0, 'setup.peer_ready_observation')
        need(re.fullmatch('[0-9a-f]{64}', p['source_sha256']) is not None, 'setup.source_hash')
        bundle = p['public_ca_bundle']
        need(bundle == peers[0]['public_ca_bundle'], 'setup.common_frozen_bundle')
        need((len(bundle) == 6) if upstream == 'http2' else (bundle == []), 'setup.trust_profile')
        need(sum(c['bytes'] for c in bundle) <= 1048576 and all(integer(c['bytes'], 1, 65536) for c in bundle), 'setup.trust_bound')
    return by_key

def request(r, inv, peers, index, kind, slot, member, completions):
    exact(r, ('epoch', 'lane', 'member', 'peer', 'generation', 'group', 'selected', 'request',
              'source_batches', 'source_inputs', 'child', 'child_source_ref', 'group_token',
              'transport_token', 'completion', 'session', 'session_monitor', 'native',
              'native_monitor', 'native_actors', 'native_reservation', 'retained_bytes',
              'original_complete', 'session_down', 'child_and_token_removed_before_source'), 'request_receipt')
    lane = ('a', 'b')[member]
    generation_index = index if kind == 'seed' else index - 1
    peer = peers[(generation_index // 8, lane)]
    need(r['member'] == member and r['lane'] == lane and r['epoch'] == generation_index // 8 and
         r['peer'] == peer['peer'] and r['generation'] == inv['generation'] and r['group'] == inv['group'], 'request.original_lineage')
    need(r['original_complete'] == 'ok' and r['session_down'] == 'normal' and
         r['child_and_token_removed_before_source'] is True, 'request.completion_attestation')
    complete = opaque(r['completion'], 'request_completion')
    need(complete not in completions, 'request.distinct_completion')
    completions.add(complete)
    q = r['request']
    expected_target = '/reload/' + kind + '/' + str(index).zfill(2) + '/' + str(slot) + '?raw=%2F'
    need(q['event'] == 'request' and q['target'] == expected_target and
         q['connection'] == generation_index % 8 + 1 and integer(q['stream'], 1) and
         [':method', 'GET'] in q['headers'] and q['ended'] is True, 'request.actual_peer_get')
    need(positive(r['retained_bytes']) and positive(r['native_reservation']), 'request.native_ledger')
    need(q['stream'] == (3 if kind == 'seed' else (7 if slot == 2 else 5)), 'request.actual_stream_sequence')
    need(r['source_inputs']['holder'] == r['native'] and integer(r['source_inputs']['deadline'], -(2**63)), 'request.source_inputs')
    need(elixir_field(r['selected'], 'resource') == inspect_id(inv['transport_controls'][member]['resource']), 'request.selected_resource')

                  'probe_monitor', 'probe_ref', 'probe_bytes', 'public_await', 'probe_down'), 'health_receipt')
        peer = peers[(idx // 8, ('a', 'b')[member])]
        need(h['resource'] == inv['transport_controls'][member]['resource'] and
             h['generation'] == inv['generation'] and h['peer'] == peer['peer'], 'health.resource_peer_lineage')
        q = h['request']
        need(q['event'] == 'request' and q['connection'] == idx % 8 + 1 and q['stream'] == 1 and
             q['target'] == '/health' and [':method', 'HEAD'] in q['headers'] and q['ended'] is True, 'health.actual_peer_head')
        need(h['probe_bytes'] == RESERVE and h['public_await'] == 'ok' and h['probe_down'] == 'normal', 'health.completion_attestation')
        cap = opaque(h['completion'], 'request_completion')
        need(cap not in completions, 'health.distinct_completion')
        completions.add(cap)

    need(ret['group'] == inv['group'] and ret['group_removed'] is True and
         ret['original_creators_down'] == 'normal' and len(ret['transports']) == 2, 'retirement.group_lineage')

        exact(row, ('resource', 'creator', 'capability', 'helper', 'helper_monitor', 'receipt',
                    'public_await', 'helper_down', 'owner_confirmed', 'charged_until_confirmation_bytes',
                    'creator_down', 'removed', 'epoch', 'lane', 'peer', 'physical'), 'retirement.transport')
        c = inv['transport_controls'][member]

             row['capability'] == c['capability'], 'retirement.original_control_lineage')

        need(key not in seen_resources, 'retirement.unique_resource')
        seen_resources.add(key)
        receipt = opaque(row['receipt'], 'retirement_receipt')
        need(receipt not in seen_receipts, 'retirement.unique_original_receipt')
        seen_receipts.add(receipt)

        need(row['lane'] == lane and row['epoch'] == idx // 8 and row['peer'] == peers[(idx // 8, lane)]['peer'], 'retirement.peer_lineage')

             row['owner_confirmed'] is True and row['removed'] is True and positive(row['charged_until_confirmation_bytes']),

def t_snapshot(s, resources, tokens=0):
    exact(s, ('status', 'independent_bytes', 'records', 'pins', 'tokens', 'monitors'), 'T_snapshot')
    need(len(s['records']) == resources and len(s['tokens']) == tokens, 'T_snapshot.table_counts')
    need(s['status']['resources'] == resources and s['status']['request_tokens'] == tokens and
         s['status']['helpers'] == 0 and s['status']['failed_resources'] == 0, 'T_snapshot.public')
    need(all(r['original_payload_bytes'] == r['independent_payload_bytes'] > 0 for r in s['records']), 'T_snapshot.payload_size')
    need(s['independent_bytes'] == sum(r['independent_payload_bytes'] for r in s['records']) +
         sum(p['independent_bytes'] for p in s['pins']) +
         sum(t['payload_bytes'] for t in s['tokens']) == s['status']['owned_serialized_payload_bytes'], 'T_snapshot.independent_sum')
    need(all(positive(t['payload_bytes']) for t in s['tokens']), 'T_snapshot.token_charge')

def g_snapshot(g, count, tokens=0):
    exact(g, ('groups', 'pins', 'tokens', 'probes', 'config_bytes', 'control_bytes', 'probe_bytes', 'limits', 'monitors'), 'G_snapshot')
    need(len(g['groups']) == count and g['limits'] == G_LIMITS and len(g['tokens']) == tokens and not g['probes'] and g['probe_bytes'] == 0, 'G_snapshot.tables')
    need(g['config_bytes'] == sum(x['installed_reserve'] for x in g['groups']), 'G_snapshot.config_sum')
    need(all(0 < x['full_record_bytes'] <= x['installed_reserve'] for x in g['groups']), 'G_snapshot.config_size')
    need(g['control_bytes'] == sum(x['control_bytes'] for x in g['pins'] + g['tokens'] + g['probes']), 'G_snapshot.controls')

def capacity(c, old, fresh, retire):
    exact(c, ('result', 'no_delta', 'holds', 'retained', 'transport_before', 'transport_after',
              'group', 'store', 'through_continue'), 'capacity')
    need(inspect_id(c['result']) == '{:error, {:capacity, :resources}}' and c['no_delta'] is True, 'capacity.public_failure')
    need(c['transport_before'] == c['transport_after'], 'capacity.exact_T_delta')
    t_snapshot(c['transport_before'], 4)
    s = c['transport_before']['status']
    need(s['pins'] == 2 and s['reserved_connections'] == 8, 'capacity.pins_connections')
    controls = old['transport_controls'] + fresh['transport_controls']
    records = {inspect_id(r['resource']): r for r in c['transport_before']['records']}
    need(set(records) == {inspect_id(r['resource']) for r in controls}, 'capacity.actual_four_resources')
    for oldc, end in zip(old['transport_controls'], retire['transports']):
        r = records[inspect_id(oldc['resource'])]
        need(r['custodian'] == oldc['custodian'] and r['capability'] == oldc['capability'] and
             r['admission'] == 'closed' and r['retirement_requested'] is True and
             r['transport_confirmed'] is False and r['custodian_down'] is False and
             end['charged_until_confirmation_bytes'] == r['original_payload_bytes'], 'capacity.old_charge_until_confirmation')
    g_snapshot(c['group'], 1)
    need(len(c['group']['pins']) == 1, 'capacity.remaining_G_pin')
    fresh_ref = elixir_field(fresh['group'], 'generation')
    old_ref = elixir_field(old['group'], 'generation')
    need(fresh_ref in inspect_id(c['group']['groups'][0]['key']) and
         all(old_ref not in inspect_id(g['key']) for g in c['group']['groups']), 'capacity.old_G_absent')
    need(c['store']['generation'] == fresh['generation'] and inspect_id(c['store']['preparation']) == 'nil' and
         inspect_id(c['store']['retired']) == '[]' and len(c['store']['baseline']) == 2, 'capacity.Store_new_only')
    need(len(c['retained']) == len(c['holds']) == len(c['through_continue']) == 2, 'capacity.paired_holds')
    need({inspect_id(r['resource']) for r in c['retained']} == {inspect_id(r['resource']) for r in old['transport_controls']}, 'capacity.retained_resources')
    ids = {inspect_id(h['id']) for h in c['holds']}
    need(len(ids) == 2 and ids == {inspect_id(h['id']) for h in c['through_continue']}, 'capacity.original_hold_ids')
    for h, clock in zip(c['holds'], c['through_continue']):
        need(h['id'] == clock['id'] and integer(h['started'], -(2**63)) and
             integer(h['transaction_elapsed_ms'], 0, 1999) and
             integer(clock['through_continue_elapsed_ms'], h['transaction_elapsed_ms'], 1999), 'capacity.original_clock_bounds')

def rollback(r, held):
    exact(r, ('result', 'prefix', 'caller', 'operation', 'original_pin_bytes', 'pin_acquired',
              'cancel_observed', 'cancellation_processed', 'no_retained_delta', 'transport_before',
              'transport_after', 'group_before', 'group_after', 'store'), 'rollback')
    need(inspect_id(r['result']) == '{:error, :group_capacity}' and positive(r['original_pin_bytes']), 'rollback.public_failure')
    need(all(r[k] is True for k in ('pin_acquired', 'cancel_observed', 'cancellation_processed', 'no_retained_delta')), 'rollback.cancellation_attestation')
    need(r['transport_before'] == r['transport_after'] and r['group_before'] == r['group_after'], 'rollback.exact_ledgers')
    calls = held['requests'] if held else []
    t_snapshot(r['transport_before'], 4, len(calls))
    g_snapshot(r['group_before'], 2, len(calls))
    tt = {inspect_id(t['ref']): t for t in r['transport_before']['tokens']}
    gt = {inspect_id(t['ref']): t for t in r['group_before']['tokens']}
    need(set(tt) == {inspect_id(c['transport_token']) for c in calls} and
         set(gt) == {inspect_id(c['group_token']) for c in calls}, 'rollback.original_held_tokens')
    for call in calls:
        t = tt[inspect_id(call['transport_token'])]
        g = gt[inspect_id(call['group_token'])]
        need(t['completion'] == g['completion'] == call['completion'] and
             inspect_id(t['resource']) == elixir_field(call['selected'], 'resource'), 'rollback.held_token_lineage')
    need(inspect_id(r['operation']) in inspect_id(r['prefix']) and ':preparation' in inspect_id(r['prefix']), 'rollback.original_prefix_operation')
    ref = elixir_field(r['prefix'], 'ref')
    need(all(inspect_id(p['ref']) != ref for p in r['transport_after']['pins']), 'rollback.acquired_prefix_absent')

    need(all(j[k] == p[k] for k in ('peer', 'address', 'port')), 'peer_join.setup_lineage')
    count = 7 if j['epoch'] == 2 else 8
    requests = (26 if j['epoch'] == 2 else 32) if profile == 'active' and j['lane'] == 'a' else (
               (20 if j['epoch'] == 2 else 24) if profile == 'active' else (14 if j['epoch'] == 2 else 16))
    need(j['joined'] == dict(event='joined', accepts=count, requests=requests, forced=0, live=0, failed=False), 'peer_join.actual_exact_counts')
    need(j['exit'] == 0 and j['collector_down'] == 'normal' and j['rebound'] is True, 'peer_join.terminal_attestation')
    need(j['resources'] == expected_resources, 'peer_join.complete_retirement_ledger')
    need([r['physical']['connection'] for r in j['resources']] == list(range(1, count + 1)), 'peer_join.actual_connection_inventory')

def audit_context(rows, key):
    contracts = [r for r in rows if r['event'].endswith('_contract')]
    setups = [r for r in rows if r['event'].endswith('_setup')]
    finished = [r for r in rows if r['event'].endswith('_finished')]
    cycles = [r for r in rows if r['event'].endswith('_cycle')]
    need(len(contracts) == len(setups) == len(finished) == 1 and len(cycles) == 22, 'context.event_counts')
    need([r['event'].rsplit('_', 1)[1] for r in rows] == ['contract', 'setup'] + ['cycle'] * 22 + ['finished'], 'context.event_order')
    contract, setup, end = contracts[0], setups[0], finished[0]
    ctx = {'event', 'upstream', 'downstream', 'profile'}
    exact(contract, ctx | {'inventory', 'outer_exunit_timeout_ms', 'embedding_lifetime_ms', 'peer_lifetime_ms',
                          'retirement_hold_ms', 'planned_generations', 'planned_transports', 'planned_replacement_cas', 'planned_heads', 'planned_gets'}, 'contract')
    for f, n in dict(outer_exunit_timeout_ms=60000, embedding_lifetime_ms=120000, peer_lifetime_ms=180000,
                     retirement_hold_ms=2000, planned_generations=23, planned_transports=46,
                     planned_replacement_cas=22, planned_heads=46, planned_gets=112 if key[2] == 'active' else 46).items():
        need(contract[f] == n, 'contract.' + f)
    need(set(contract['inventory']['plateau']) == PLATEAU, 'contract.plateau_fields')
    exact(setup, ctx | {'identity', 'entered_monotonic_ms', 'setup_elapsed_ms', 'embedding_creation_interval_monotonic_ms',
                       'peers', 'classes', 'ordered_pair_class', 'initial'}, 'setup')
    need(integer(setup['setup_elapsed_ms'], 0, 59999), 'setup.clock')
    peers = peer_setup(setup, key[0])
    need([r['cycle'] for r in cycles] == list(range(1, 23)), 'cycles.exact_sequence')
    warmup2 = cycles[1]['vector']
    controls = []
    ledgers = [setup['initial']] + [c['ledger'] for c in cycles]
    identities = set()
    completions, resources, receipts = set(), set(), set()
    joins = []
    retired_by_peer = {inspect_id(p['peer']): [] for p in setup['peers']}
    for idx, ledger in enumerate(ledgers):
        inv = ledger['inventory']
        inventory(inv, setup['identity'], ledgers[idx-1]['inventory'] if idx else None)
        need(canonical(inv['generation']) not in identities, 'generation.distinct')
        identities.add(canonical(inv['generation']))
        health(ledger['health'], inv, peers, idx, completions)
        need(len(ledger['seed']) == 2, 'seed.two')
        for slot, r in enumerate(ledger['seed']):
            request(r, inv, peers, idx, 'seed', slot, slot, completions)
        if idx == 0:
            exact(ledger, ('index', 'initial_publication', 'health', 'seed', 'inventory'), 'initial')
            need(ledger['index'] == 0 and ledger['initial_publication']['activated'] == inv['generation'], 'initial.CAS')
            continue
        c = cycles[idx-1]
        exact(c, ctx | {'cycle', 'measured', 'vector', 'ledger', 'elapsed_ms'}, 'cycle')
        exact(ledger, ('index', 'cas', 'health', 'seed', 'held', 'retirement', 'rollback', 'capacity', 'epoch_joins', 'inventory'), 'cycle.ledger')
        need(ledger['index'] == idx and c['measured'] is (idx >= 3) and integer(c['elapsed_ms'], 0, 59999), 'cycle.index_clock')
        old = ledgers[idx-1]['inventory']
        need(ledger['cas'] == dict(expected=old['generation'], activated=inv['generation']), 'CAS.exact_lineage')
        vector(c['vector'], setup['classes'], setup['ordered_pair_class'])
        if idx >= 3:
            need(c['vector'] == warmup2, 'plateau.full_vector_equal_warmup2')
        if key[2] == 'active':
            held = ledger['held']
            need(held['last_selected_member'] == 0 and held['retained_unselected_member'] == 1 and len(held['requests']) == 3, 'active.last_source')
            for slot, (r, member) in enumerate(zip(held['requests'], (0, 1, 0))):
                request(r, old, peers, idx, 'held', slot, member, completions)
            need(held['original_sources'] == held['requests'][-1]['source_batches'], 'active.original_last_source')
        else:
            need(ledger['held'] is None, 'idle.no_held_requests')
        rr = retirement(ledger['retirement'], old, peers, idx-1, resources, receipts)
        for r in rr:
            retired_by_peer[inspect_id(r['peer'])].append(r)
        if idx == 3:
            rollback(ledger['rollback'], ledger['held'])
            capacity(ledger['capacity'], old, inv, ledger['retirement'])
        else:
            need(ledger['rollback'] is None and ledger['capacity'] is None, 'cycle.single_capacity_rollback')
        need(len(ledger['epoch_joins']) == (2 if idx in (8, 16) else 0), 'epoch.join_schedule')
        joins.extend(ledger['epoch_joins'])
    exact(end, ctx | {'measured_samples', 'totals', 'final_retirement', 'final_epochs', 'final_zero', 'teardown_elapsed_ms', 'profile_elapsed_ms'}, 'finished')
    final_zero(end['final_zero'], setup['identity'])
    need(end['measured_samples'] == 20 and integer(end['profile_elapsed_ms'], 0, 59999) and
         integer(end['teardown_elapsed_ms'], 0, end['profile_elapsed_ms']), 'finished.samples_clocks')
    final = retirement(end['final_retirement'], ledgers[-1]['inventory'], peers, 22, resources, receipts)
    for r in final:
        retired_by_peer[inspect_id(r['peer'])].append(r)
    need(len(end['final_epochs']) == 2, 'finished.final_epochs')
    joins.extend(end['final_epochs'])
    need(len(joins) == 6 and {(j['epoch'], j['lane']) for j in joins} == set(peers), 'peer_joins.six')
    for j in joins:
        join_peer(j, peers, retired_by_peer[inspect_id(j['peer'])], key[2])
    totals = dict(generations=23, distinct_transports=46, initial_publication=1, replacement_cas=22,
                  heads=46, seed_gets=46, active_gets=66 if key[2] == 'active' else 0,
                  gets=112 if key[2] == 'active' else 46, requests=158 if key[2] == 'active' else 92,
                  retirement_receipts=46, physical_closes=46, group_removals=23, peer_joins=6, accepts=46)
    need(end['totals'] == totals and len(resources) == len(receipts) == 46, 'finished.derived_totals')
    need(len(completions) == totals['requests'], 'finished.original_completions')
    return totals

def receipt_audit(receipt, raw_log, candidate):
    need(candidate['kind'] == 'GROUP_RELOAD_SOURCE_CANDIDATE2' and
         candidate['base_commit'] == 'e4a7660baa61a792f4245f511b190a0fcfcbf920' and
         len(candidate['actual_command_source_sha256']) == candidate['actual_command_source_count_with_enum'] == 402,
         'candidate.source_map_shape')
    expected_sources = dict(candidate['actual_command_source_sha256'])
    excluded_enum_sha256 = expected_sources.pop(EXCLUDED_ENUM)
    need(excluded_enum_sha256 == '42ad21b2f1d4a317a00a33e90712bc7aa19790a812ce06984fce5e0355eff0ca' and
         len(expected_sources) == 401, 'candidate.exact_collector_map_without_enum')
    need(receipt['name'] == candidate['first_command']['name'] and receipt['cwd'] == ROOT and
         receipt['environment'] == {'MIX_ENV': 'test', 'ERL_FLAGS': '+S 2:2'} and
         receipt['browser'] is None and receipt['log'] == Path(candidate['first_command']['log']).name,
         'command.exact_context')
    collector = candidate['first_command']['outer_argv'][2]
    need(receipt['collector_sha256'] == candidate['actual_command_source_sha256'][collector], 'command.collector_hash')
    need(receipt['kind'] == 'green' and receipt['verdict'] == 'PASS' and receipt['actual_exit'] == 0 and
         receipt['forced'] is False and receipt['wait_error'] is None and receipt['process_joined'] is True and
         receipt['process_group_remaining'] == [], 'command.actual_joined_nonforced')
    need(receipt['source_stable'] is True and receipt['source_changes'] == [] and
         receipt['source_before'] == receipt['source_after'], 'command.source_stability')
    need(receipt['source_before'] == receipt['source_after'] == expected_sources and
         receipt['source_before'][SOURCE] == SOURCE_SHA, 'command.exact_frozen_input_map')
    need(receipt['command'] == candidate['first_command']['payload_argv'], 'command.exact_frozen_argv')
    need(receipt['log_sha256'] == hashlib.sha256(raw_log).hexdigest(), 'command.log_hash')
    need(positive(receipt['tests']) and receipt['reported_tests'] == receipt['tests'] and
         all(receipt[k] == 0 for k in ('failures', 'skipped', 'excluded')) and receipt['warnings'] == [], 'command.nonzero_clean_tests')
    dc = receipt['descendant_cleanup']
    need(dc['forced'] is False and dc['remaining'] == [] and dc['signal_attempts'] == [] and
         dc['reap_deadline_exhausted'] == [] and dc['initial_after_reap'] == [], 'command.actual_descendant_joins')
    waits = []
    for row in dc['reaped']:
        need(integer(row['pid'], 1) and integer(row['wait_status'], 0, 65535), 'command.raw_wait')
        # Preserve genuine signal termination, including SIGUSR1(-10), unchanged.
        waits.append(dict(row, decoded_exit=os.waitstatus_to_exitcode(row['wait_status'])))
    return waits

def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--receipt', required=True)
    p.add_argument('--log', required=True)
    p.add_argument('--candidate', required=True)
    a = p.parse_args()
    result = dict(kind='PROPOSED_GROUP_RELOAD_RUNTIME_DATA_AUDIT3', status='FAIL',
                  runtime_executed_by_audit=False, source_sha256=SOURCE_SHA, contexts=0,
                  source_attestations=[], counts_separate='ExUnit only; Python/browser/static excluded')
    try:
        candidate_bytes = bounded(a.candidate, RECEIPT_MAX)
        need(hashlib.sha256(candidate_bytes).hexdigest() == CANDIDATE_SHA, 'candidate.original_frozen_hash')
        candidate = parse_json(candidate_bytes)
        result['excluded_collector_source'] = dict(path=EXCLUDED_ENUM,
             sha256=candidate['actual_command_source_sha256'][EXCLUDED_ENUM],
             reason='Explicit frozen enum JSON is outside the collector E6 .py/.exs whitelist.')
        result['expected_collector_source_count'] = 401
        result['candidate_source_count_with_enum'] = 402
        receipt_bytes = bounded(a.receipt, RECEIPT_MAX)
        receipt = parse_json(receipt_bytes)
        raw_log = bounded(a.log, LOG_MAX)
        result['actual_raw_waits'] = receipt_audit(receipt, raw_log, candidate)
        plain = re.sub(r'\x1b\[[0-9;]*[A-Za-z]', '', raw_log.decode('utf-8', errors='strict'))
        summaries = re.findall(r'(\d+) tests?, (\d+) failures?([^\n]*)', plain)
        need(summaries and sum(int(t) for t, _, _ in summaries) == receipt['tests'] and
             sum(int(f) for _, f, _ in summaries) == 0 and
             not re.search(r'\b[1-9]\d* (skipped|excluded)\b', plain) and
             not any('warning:' in line and not line.startswith('warning: Git tree ') for line in plain.splitlines()), 'log.actual_test_summary')
        grouped = {key: [] for key in CONTEXTS}
        for line in plain.splitlines():
            need(len(line.encode('utf-8')) <= LINE_MAX, 'log.line_bound')
            if '"group_h2_reload_' not in line:
                continue
            start = line.find('{')
            need(start >= 0, 'log.event_json_start')
            row = parse_json(line[start:])
            event = row.get('event')
            need(event in {'group_h2_reload_' + e for e in EVENTS}, 'log.event_name')
            key = (row.get('upstream'), row.get('downstream'), row.get('profile'))
            need(key in CONTEXTS, 'log.exact_context')
            grouped[key].append(row)
            need(sum(map(len, grouped.values())) <= 200, 'log.event_count_bound')
        need(sum(map(len, grouped.values())) == 200, 'log.exact_200_events')
        totals = [audit_context(rows, key) for key, rows in sorted(grouped.items())]
        aggregate = {k: sum(row[k] for row in totals) for k in totals[0]}
        expected = dict(generations=184, distinct_transports=368, initial_publication=8,
                        replacement_cas=176, heads=368, seed_gets=368, active_gets=264,
                        gets=632, requests=1000, retirement_receipts=368,
                        physical_closes=368, group_removals=184, peer_joins=48, accepts=368)
        need(aggregate == expected, 'aggregate.original_ledger_counts')
        result.update(contexts=8, aggregate=aggregate, emitted_data_checks='PASS',
                      receipt_sha256=hashlib.sha256(receipt_bytes).hexdigest(), log_sha256=hashlib.sha256(raw_log).hexdigest())
        result['final_zero_contexts_checked'] = 8
        result['source_attestations'].append(dict(
             operations=['original completion awaits', 'normal monitor waits', 'processed cancellation',
                         'ETF byte measurements', 'absence of unexpected peer effects'],
             detail='Frozen tests assert these operations; audit checks emitted identities, flags, charges and peer frames. It does not replay source assertions.'))
        result['status'] = 'PASS'
        print(json.dumps(result, indent=2, sort_keys=True))
        return 0
    except (AuditError, KeyError, IndexError, AttributeError, TypeError, ValueError,
            OSError, OverflowError, RecursionError) as error:
        result['failure'] = str(error)[:512]
        print(json.dumps(result, indent=2, sort_keys=True))
        return 1

if __name__ == '__main__':
    sys.exit(main())
