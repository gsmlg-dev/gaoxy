def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load(name):
    return json.loads((E / name).read_bytes())


def plain_log(path):
    return re.sub(r'\x1b\[[0-9;]*[A-Za-z]', '', path.read_text())


def summary(plain):
    summaries = re.findall(r'(\d+) tests?, (\d+) failures?([^\n]*)', plain)
    reported = sum(int(row[0]) for row in summaries)
    failures = sum(int(row[1]) for row in summaries)
    skipped = sum(int(n) for n in re.findall(r'(\d+) skipped', plain))
    excluded = sum(int(n) for n in re.findall(r'(\d+) excluded', plain))
    return reported, failures, skipped, excluded


def completed_cases(plain, names):
    rows = []
    lines = plain.splitlines()
    for name in names:
        pattern = re.compile(r'\s*\* ' + re.escape(name) +
                             r' \([0-9.]+ms\) \[L#[0-9]+\]')
        matches = [line for line in lines if pattern.fullmatch(line)]
        assert len(matches) == 1, (name, matches)
        rows.append({'name': name, 'completed_trace_line': matches[0]})
    return rows
